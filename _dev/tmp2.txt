3. Frontend Helpers:
The exact source code for makeOptions (usually in front/src/lib/index.js or $lib).
(To prove how relational foreign keys are transformed into DOM options).
===============
/**
 *
 * @param raw
 * @param field
 * @returns {{lit: *[], type: *[]}}
 */
export const makeOptions = (raw, field = 'name') => {
    if (!Array.isArray(raw)) raw = Object.entries(raw)
    let data = {type: [], lit: []}
    raw.forEach(o => {
        data.type.push(o.id)
        data.lit.push(o[field] || o.id)
    })
    return data;
}
