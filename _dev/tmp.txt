export const collectFormData = (target = 'form.r2wx-form') => {

    let form = document.querySelector(target),
        formData = new FormData(form);

    let data = {};

    [...formData].forEach(
        ([key, val]) => {

            if (val === '\n') return;

            if (['true', 'false'].includes(val)) val = (val === 'true');

            if (parseInt(val) || parseFloat(val)) val = +val;

            if (key in data) {
                data[key] = Array.isArray(data[key]) ? [...data[key], val] : [data[key], val]
                return data
            }

            data[key] = val
        }
    )

    /*
    COLLECT 'DATEPICKERS' (CHECKBOXES **)
     */
    let bools = [...document.querySelectorAll('input[type=checkbox]')]
        .map(elem => [elem.name, elem.value])

    bools.forEach(
        ([id, val]) => data[id] = val
    )

    /*
    COLLECT 'DATEPICKERS' (DATE **)
     */
    let dates = [...document.querySelectorAll(".r2wx-proxy-id")]
        .map(o => (
            [o.id, o.querySelector('input').value])
        )

    dates.forEach(
        ([id, val]) => data[id] = val.split("-").reverse().join("-")
    )

    return data;
}

}

import _authall from "$lib/server/authall.js";
import {_reply} from "$lib/server/index.js";
import httpServer from "$lib/httpServer.js";
import {ROLE} from "$var";


export async function POST(ctx) {
    try {

        const Auth = _authall();
        if (!Auth.accessRole(ctx, ROLE.GLOB)) return _reply({error: 'NOROL'});

        const path = [
            ctx.params.tool_id,
        ].join("/")

        const data = await ctx.request.json();
        console.log(" -- POST:: PATH / DATA:: ", {path, data})
        const res = await httpServer.post(path, data)

        console.log(" -- POST:: RES:: ", res)
        return _reply(res);

    } catch (error) {
        console.log("++++++++++++++++++++++", error)
        throw new Error(error);
    }
}

model TOOL_SAMX {
  id Int @id @default(autoincrement())

  name        String  @db.VarChar(160)
  description String? @db.Text

  status   ENUM_STATUS   @default(DRAFT)
  priority ENUM_PRIORITY @default(NORMAL)

  effort   Float?
  progress Int    @default(0)

  starts_at DateTime?
  due_at    DateTime?

  billable Boolean @default(false)
  active   Boolean @default(true)

  category_id  Int
  StatCategory STAT_CATEGORY @relation(fields: [category_id], references: [id])

  owner_id Int?
  Owner    OWNER? @relation(fields: [owner_id], references: [id])

  created DateTime @default(now())
  updated DateTime @updatedAt
}

model STAT_CATEGORY {
  id     Int     @id @default(autoincrement())
  name   String  @db.VarChar(80)
  active Boolean @default(true)

  TOOL_SAMX TOOL_SAMX[]
}

model OWNER {
  id     Int     @id @default(autoincrement())
  name   String  @db.VarChar(120)
  active Boolean @default(true)

  TOOL_SAMX TOOL_SAMX[]
}

enum ENUM_STATUS {
  DRAFT
  READY
  ACTIVE
  PAUSED
  CLOSED
}

enum ENUM_PRIORITY {
  LOW
  NORMAL
  HIGH
  CRITICAL
}
